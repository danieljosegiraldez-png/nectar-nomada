async function seedMeliponiniCatalog(databaseClient) {
  console.log("🚀 Starting Néctar Nómada Meliponini Catalog Seeding...");
  
  try {
    // Example abstraction for Knex, Prisma, or Mongoose execution
    for (const species of meliponiniCatalogData) {
      console.log(`Inserting taxonomy payload for: ${species.scientificName}`);
      
      // Replace with your ORM syntax: e.g., await databaseClient.species.upsert({ ... })
      await databaseClient("species_catalog")
        .insert({
          id: species.id,
          scientific_name: species.scientificName,
          local_names: JSON.stringify(species.localNames),
          genus: species.genus,
          distribution_panama: JSON.stringify(species.distributionPanama),
          morphology: JSON.stringify(species.morphology),
          ecology: JSON.stringify(species.ecology),
          meliponiculture_specs: JSON.stringify(species.meliponicultureSpecs),
          management_protocols: JSON.stringify(species.managementProtocols),
          created_at: new Date(),
          updated_at: new Date()
        })
        .onConflict("id")
        .merge();
    }
    
    console.log("✅ Meliponini Catalog seed completed successfully.");
  } catch (error) {
    console.error("❌ Seeding execution failure:", error);
    throw error;
  }
}
