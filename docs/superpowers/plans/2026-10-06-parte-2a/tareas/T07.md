/**
 * El resultado de una promesa sin que llegue a rechazarse nunca (copiado de `corridaConProceso.test.ts`): mientras la prueba
 * espera a otra cosa, un registro que falla no deja un rechazo sin atender.
 */
type Resultado<T> = { ok: true; valor: T } | { ok: false; error: unknown };
const resultadoDe = <T>(p: Promise<T>): Promise<Resultado<T>> =>
  p.then((v) => ({ ok: true as const, valor: v }), (error: unknown) => ({ ok: false as const, error }));

/**
 * Una transacción AJENA que hace `trabajo` y se queda abierta hasta que se la suelte (copiado de `corridaConProceso.test.ts`).
 * `pid` es su proceso en la base, para preguntarle a `pg_blocking_pids` quién la espera. **Hay que llamar siempre a
 * `soltar()`, también si la prueba falla (en un `finally`)**: si no, el `afterAll` esperaría a lo que retiene.
 */
function retener(trabajo: (tx: Prisma.TransactionClient) => Promise<unknown>) {
  let soltar!: () => void;
  const suelta = new Promise<void>((resolver) => { soltar = resolver; });
  let tomado!: (pid: number) => void;
  let fallo!: (error: unknown) => void;
  const pid = new Promise<number>((resolver, rechazar) => { tomado = resolver; fallo = rechazar; });
  const hecho = prisma.$transaction(async (tx) => {
    try {
      await trabajo(tx);
      const [fila] = await tx.$queryRaw<{ pid: number }[]>`SELECT pg_backend_pid() AS pid`;
      tomado(fila!.pid);
    } catch (error) {
      fallo(error);
      throw error;
    }
    await suelta;
  }, { timeout: 30000, maxWait: 10000 });
  hecho.catch(() => undefined); // quien la espera la ve rechazarse al `await hecho`; esto sólo evita el rechazo sin atender
  return { pid, soltar, hecho };
}

/** Se OBSERVA en la base que alguien espera a la sesión `pid`, en vez de inferirlo de que algo tarda. */
async function esperarQueAlguienEspere(pid: number, mensaje: string) {
  let esperando = 0;
  for (let intento = 0; intento < 100 && esperando === 0; intento++) {
    const [fila] = await prisma.$queryRaw<{ n: number }[]>`
      SELECT count(*)::int AS n FROM pg_stat_activity WHERE ${pid}::int = ANY(pg_blocking_pids(pid))`;
    esperando = fila!.n;
    if (esperando === 0) await new Promise((r) => setTimeout(r, 50));
  }
  expect(esperando, mensaje).toBeGreaterThan(0);
}

    // corrida en crudo. Todavía no confirma: quien lea ahora la ve abierta. Sin el bloqueo, el manejo no espera a nadie —su
    // clave foránea a la corrida (`FOR KEY SHARE`) no choca con la actualización de `ended_at` (`FOR NO KEY UPDATE`)—.
    const termina = retener(async (tx) => {

import { LotProcessError } from "../traceability/errorDeProceso";
import { TIPOS_POR_REGISTRO, type RegistroDePaso } from "./vocabulario";

export async function exigePasoCoherente(
  tx: Prisma.TransactionClient,
  input: {
    registro: RegistroDePaso;
    recipeVersionIdDelProceso: string | null;
    recetaTienePasos: boolean;
    recipeStepId: string | null;
    stepTypeValueId: string | null;
    motivoDesviacion: string | null;
  },
): Promise<{ recipeStepId: string | null; stepTypeValueId: string | null; motivoDesviacion: string | null }> {
  const cumple: readonly string[] = TIPOS_POR_REGISTRO[input.registro];

  if (input.recipeStepId !== null) {
    const paso = await tx.processRecipeStep.findUnique({
      where: { id: input.recipeStepId },
      select: { id: true, recipeVersionId: true, stepTypeValueId: true },
    });
    if (!paso || paso.recipeVersionId !== input.recipeVersionIdDelProceso) throw new LotProcessError("paso_de_otra_receta");
    const tipo = await tx.variableCatalogValue.findUniqueOrThrow({ where: { id: paso.stepTypeValueId }, select: { value: true } });
    if (!cumple.includes(tipo.value)) throw new LotProcessError("paso_no_corresponde");
    if (input.stepTypeValueId !== null && input.stepTypeValueId.toLowerCase() !== paso.stepTypeValueId) {
      throw new LotProcessError("paso_no_corresponde");
    }
    return { recipeStepId: paso.id, stepTypeValueId: paso.stepTypeValueId, motivoDesviacion: null };
  }

  if (input.stepTypeValueId !== null) {
    const declarado = await tx.variableCatalogValue.findUnique({
      where: { id: input.stepTypeValueId },
      select: { value: true, catalog: { select: { key: true } } },
    });
    if (!declarado || declarado.catalog.key !== "tipo_paso" || !cumple.includes(declarado.value)) {
      throw new LotProcessError("paso_no_corresponde");
    }
  }
  const esDesviacion = input.recipeVersionIdDelProceso !== null && input.recetaTienePasos;
  const motivo = input.motivoDesviacion?.trim() || null;
  if (esDesviacion && motivo === null) throw new LotProcessError("desviacion_sin_motivo");
  return { recipeStepId: null, stepTypeValueId: input.stepTypeValueId, motivoDesviacion: esDesviacion ? motivo : null };
}

  motivoDesviacion?: string | null;

export async function recordFermentationIntervention(userAccountId: string, input: RecordFermentationInterventionInput) {
  const sourceLot = await resolveRunSourceLot(input.fermentationRunId);
  await requireLotAccess(userAccountId, "manage", [{ projectId: sourceLot.projectId, locationId: sourceLot.locationId, classification: sourceLot.classification }]);

  // Parte 2a (2026-10-03; registro del plan: «deciden DENTRO de una transacción con bloquearLinaje»). Escribía sin
  // transacción, sin AuditEvent y sin mirar si la corrida seguía abierta. Ahora todo va dentro, DESPUÉS de bloquear el linaje
  // del lote de la corrida —el mismo que bloquean empezarla, cerrar su proceso y dividirlo—, y la corrida se lee ya con él
  // bloqueado.
  return prisma.$transaction(async (tx) => {
    await bloquearLinaje(tx, sourceLot.id);
    const corrida = await tx.fermentationRun.findUniqueOrThrow({
      where: { id: input.fermentationRunId },
      select: { endedAt: true, processRecipeVersionId: true },
    });
    if (corrida.endedAt !== null) throw new TraceabilityAccessError("fermentation_run_already_ended");

    // §4.1: sólo una inoculación o una adición cumple un paso. Su tipo de paso ES su tipo de intervención —los dos vocabularios
    // llaman igual a esos dos actos: inferencia de la casa, no del paquete—, así que una adición no cumple un paso de
    // inoculación. La versión es la de la corrida: desde R4 es la de su proceso, y con la corrida abierta ese proceso sigue
    // abierto (R5: no se cierra ni se divide con corridas abiertas).
    const cumplen: readonly string[] = TIPOS_POR_REGISTRO.fermentationIntervention;
    const cumpleUnPaso = cumplen.includes(input.interventionType);
    if (!cumpleUnPaso && input.recipeStepId) throw new LotProcessError("paso_no_corresponde");
    const paso = cumpleUnPaso
      ? await exigePasoCoherente(tx, {
          registro: "fermentationIntervention",
          recipeVersionIdDelProceso: corrida.processRecipeVersionId,
          recetaTienePasos:
            corrida.processRecipeVersionId !== null &&
            (await tx.processRecipeStep.count({ where: { recipeVersionId: corrida.processRecipeVersionId } })) > 0,
          recipeStepId: input.recipeStepId ?? null,
          stepTypeValueId: (
            await tx.variableCatalogValue.findFirstOrThrow({
              where: { value: input.interventionType, catalog: { key: "tipo_paso" } },
              select: { id: true },
            })
          ).id,
          motivoDesviacion: input.motivoDesviacion ?? null,
        })
      : { recipeStepId: null, stepTypeValueId: null, motivoDesviacion: null };

    const intervencion = await tx.fermentationIntervention.create({
      data: {
        fermentationRunId: input.fermentationRunId,
        interventionType: input.interventionType,
        occurredAt: input.occurredAt,
        notes: input.notes ?? null,
        stepTypeValueId: paso.stepTypeValueId,
        recipeStepId: paso.recipeStepId,
        motivoDesviacion: paso.motivoDesviacion,
        createdBy: userAccountId,
      },
    });

    await recordAuditEvent(
      {
        actorUserAccountId: userAccountId,
        operation: "fermentation_intervention.record",
        entityType: "fermentation_intervention",
        entityId: intervencion.id,
        after: intervencion,
        sourceInterface: "traceability.service",
      },
      tx,
    );

    return intervencion;
  }, TRANSACCION_DEL_LINAJE);
}
