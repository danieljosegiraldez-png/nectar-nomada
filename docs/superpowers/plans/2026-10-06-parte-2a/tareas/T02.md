  note?: string;
  tiposDePaso: readonly TipoDePaso[];
}

// TypeScript sólo infiere del JSON una unión de formas; la forma real la vigila
// `tests/recetas/referencias.test.ts` («cada parámetro trae fuente y confianza…»).
const PARAMETROS = (copia as unknown as { parametros: readonly ParametroCopiado[] }).parametros;

/**
 * Las fuentes que son un fabricante hablando de su producto, según `_meta.sources` del paquete: «Fermentis
 * SafCoffee product pages (2026)» y «Penagos equipment pages (vendor)».
 */
const FUENTES_DE_FABRICANTE: readonly string[] = ["FERMENTIS", "PENAGOS"];

function fuentesDe(p: ParametroCopiado): readonly string[] {
  return typeof p.source === "string" ? [p.source] : p.source;
}

/** El id de una fuente es su primera palabra: «CENICAFE AT577» es CENICAFE; «NN gate» es NN. */
function idDeFuente(fuente: string): string {
  return fuente.trim().split(/\s+/)[0] ?? "";
}

/** Un valor del paquete como texto: un par de números es un rango; un objeto, sus sub-valores con nombre. */
function texto(v: unknown): string {
  if (Array.isArray(v)) {
    return v.length === 2 && v.every((x) => typeof x === "number") ? `${v[0]}–${v[1]}` : v.map(texto).join(", ");
  }
  if (v !== null && typeof v === "object") {
    return Object.entries(v)
      .map(([nombre, x]) => `${nombre}: ${texto(x)}`)
      .join(" · ");
  }
  return String(v);
}

function conUnidad(base: string, unidad: string | undefined): string {
  return unidad ? `${base} ${unidad}` : base;
}

/** Las formas en que el paquete da un valor: valor, valor con tope, rango, o un solo extremo. */
function valorDe(p: ParametroCopiado): string {
  const { value, min, max } = p;
  let base: string;
  if (value !== undefined && max !== undefined && min === undefined) base = `${texto(value)} (≤ ${max})`;
  else if (value !== undefined) base = texto(value);
  else if (min !== undefined && max !== undefined) base = `${min}–${max}`;
  else if (max !== undefined) base = `≤ ${max}`;
  else if (min !== undefined) base = `≥ ${min}`;
  else base = "—";
  return conUnidad(base, p.unit);
}

function aReferencia(p: ParametroCopiado, tipo: TipoDePaso): ReferenciaDelPaquete {
  const fuentes = fuentesDe(p);
  return {
    parametro: p.key,
    tipoDePaso: tipo,
    valor: valorDe(p),
    fuente: fuentes.join(" + ") + (p.profile ? ` · perfil ${p.profile}` : ""),
    confianza: p.confidence,
    alternativas: Object.entries(p.alt ?? {}).map(([autoridad, v]) => ({ valor: conUnidad(texto(v), p.unit), fuente: autoridad })),
    marcarVisible: p.confidence === "low" || fuentes.some((f) => idDeFuente(f) === "NN"),
    noPrecargar:
      p.scope !== undefined || p.use !== undefined || fuentes.some((f) => FUENTES_DE_FABRICANTE.includes(idDeFuente(f))),
    nota: p.note ?? null,
  };
}

 * - `backslopped` es el valor del eje D del paquete (`processing_axes.json`, `D_microbial_control`: «own
 *   must/mosto/lixiviado from a previous batch»): lo que la casa guarda como `doble_mosto` de `sustrato_anadido`, y
 *   cuyos otros nombres (`mosto`, `mossto`, `lixiviado`, «previous-batch starter») son alias de ese mismo valor dentro de su
 *   catálogo, y los siembra la tarea 8 del plan (no esta lista: aquí sólo se muestran, no se guardan ni se leen).
 * - «atomizado» NO está aquí: es un método de inoculación (`metodo_inoculacion`,

export const SINONIMOS_ENTRE_CATALOGOS: readonly { termino: string; catalogo: string; valor: string }[] = [
  { termino: "mosto propio", catalogo: "medio_lavado", valor: "mosto_propio" },

const POR_TIPO = new Map<TipoDePaso, ReferenciaDelPaquete[]>();
for (const p of PARAMETROS) {

-- contar-semi-lavados.sql — SÓLO LECTURA (Parte 2a, tarea 2; registro, Ruling M3).
--
-- Para qué. «Semi Wash 50%» y «Semi Wash 75%» se leían como el porcentaje de mucílago QUITADO (ADR-181, punto 12) y Daniel
-- decidió el 2026-10-03 que son lo que QUEDA (corregido el 2026-10-04). Con el 50 % las dos lecturas coinciden; con el 75 %
-- no: antes quedaba el 25 %, ahora el 75 %. Esta consulta cuenta qué hay YA registrado con esos dos valores, para que
-- Daniel decida cómo se leen los viejos ANTES del reimport.
--
-- Qué es. UN solo SELECT, sin escrituras: se pega tal cual en el SQL Editor de Neon o se corre con psql. Los valores se
-- buscan por su TEXTO (`value`) dentro del catálogo `grado_proceso`, nunca por id: los ids cambian de una base a otra.
--
-- Cómo se lee. Tres columnas, `seccion`, `detalle` y `n`:
--   0 · base                     a qué base habló: se lee ANTES que nada.
--   1 · procesos por grado       los cinco valores del catálogo, CON ceros. Natural, Washed y Honey no son lo que se busca:
--                                son lo que demuestra que la consulta mide algo (un cero en los dos semi-lavados, junto a
--                                un número en los demás, es un cero medido).
--   2 · control                  procesos cuyo grado no es de este catálogo: tiene que ser 0.
--   3 · procesos con Semi Wash   uno por fila: grado, lote, organización, número de proceso y fecha.
--   4 · otras columnas           CUALQUIER columna de la base que apunte a un valor de ese catálogo (p. ej. los
--                                tratamientos de Research OS): sólo lo que no es cero, para los cinco valores.
--   5 · control de la sección 4  lo que la sección 4 cuenta en `lot_process` contra lo que cuenta la sección 1, y que la 4
--                                haya mirado esa columna con los cinco valores: son dos formas de contar lo mismo y tienen
--                                que dar 0 de diferencia. Si la 4 no encontrara la columna, daría 5, no 0.
-- Dónde se guarda el grado de un proceso: `traceability.lot_process.process_grade_value_id`, clave ajena obligatoria hacia
-- `research.variable_catalog_value`.

with
catalogo as (
  select v.id, v.value, v.display_order
  from research.variable_catalog c
  join research.variable_catalog_value v on v.catalog_id = c.id
  where c.key = 'grado_proceso'
),
por_grado as (
  select g.display_order, g.value, count(p.id) as procesos
  from catalogo g
  left join traceability.lot_process p on p.process_grade_value_id = g.id
  group by g.display_order, g.value
),
semi as (
  select g.value as grado, l.lot_code, coalesce(o.name, '(sin organización)') as organizacion, p.sequence_order, p.started_at
  from traceability.lot_process p
  join catalogo g on g.id = p.process_grade_value_id
  join traceability.lot l on l.id = p.lot_id
  left join core.organization o on o.id = l.organization_id
  where g.value in ('Semi Wash 50%', 'Semi Wash 75%')
),
referencias as (
  select con.conrelid::regclass::text as tabla, att.attname::text as columna
  from pg_constraint con
  join pg_attribute att on att.attrelid = con.conrelid and att.attnum = con.conkey[1]
  where con.contype = 'f'
    and con.confrelid = 'research.variable_catalog_value'::regclass
    and array_length(con.conkey, 1) = 1
),
cuentas as (
  select r.tabla, r.columna, g.value as grado,
         (xpath('/row/n/text()', query_to_xml(
            format('select count(*) as n from %s t where t.%I = %L::uuid', r.tabla, r.columna, g.id), false, true, '')))[1]::text::bigint as filas
  from referencias r
  cross join catalogo g
)
select '0 · base' as seccion, current_database()::text as detalle, null::bigint as n
union all
select '1 · procesos de lote por grado (todos los valores del catálogo, con ceros)', value, procesos from por_grado
union all
select '2 · control: procesos con un grado que NO es de este catálogo (tiene que ser 0)', 'lot_process',
       ((select count(*) from traceability.lot_process) - (select coalesce(sum(procesos), 0) from por_grado))::bigint
union all
select '3 · procesos con Semi Wash 50 % o 75 %',
       grado || ' · ' || lot_code || ' · ' || organizacion || ' · proceso ' || sequence_order || ' · ' || to_char(started_at, 'YYYY-MM-DD'),
       1::bigint
from semi
union all
select '4 · columnas que apuntan a un valor de grado_proceso (sólo lo que no es cero)', tabla || '.' || columna || ' · ' || grado, filas
from cuentas
where filas > 0
union all
select '5 · control: lot_process contado por la sección 4 y por la 1, diferencia en filas y en valores (tiene que ser 0)', 'lot_process.process_grade_value_id',
       (select abs(coalesce(sum(c.filas), 0) - (select coalesce(sum(procesos), 0) from por_grado))
               + abs(count(*) - (select count(*) from por_grado))
        from cuentas c
        where c.tabla in ('traceability.lot_process', 'lot_process') and c.columna = 'process_grade_value_id')::bigint
order by 1, 2;
