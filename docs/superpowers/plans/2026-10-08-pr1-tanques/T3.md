function retenerRetirando(equipmentId: string) {

  const suelta = new Promise<void>((resolver) => {

  let tomado!: (pid: number) => void;

  const pid = new Promise<number>((resolver, rechazar) => {
    tomado = resolver;
    fallo = rechazar;
  });
  const hecho = prisma.$transaction(
    async (tx) => {
      try {
        await tx.$queryRaw`SELECT 1 FROM "core"."equipment" WHERE "id" = ${equipmentId}::uuid FOR UPDATE`;
        await tx.equipment.update({
          where: { id: equipmentId },
          data: { lifecycleStatus: "retired", retiredAt: new Date(), retiroMotivo: "TEST: el retiro que llega a mitad de camino" },
        });
        const [fila] = await tx.$queryRaw<{ pid: number }[]>`SELECT pg_backend_pid() AS pid`;
        tomado(fila!.pid);
      } catch (error) {
        fallo(error);
        throw error;
      }
      await suelta;
    },
    { timeout: 30000, maxWait: 10000 },
  );
  hecho.catch(() => undefined); // quien la espera la ve rechazarse en `await hecho`; esto sólo evita el rechazo sin atender
  return { pid, soltar, hecho };

    select: { organizationId: true, projectId: true, locationId: true, classification: true },
  });

  await requireLotAccess(userAccountId, "view", [{ projectId: lote.projectId, locationId: lote.locationId, classification: lote.classification }]);
  const equipos = await prisma.equipment.findMany({
    where: dondeCandidatoAFermentar(lote.organizationId),

      id: true,
      name: true,
      projectId: true,
      classification: true,
      lifecycleStatus: true,

      fermentationRuns: { where: { endedAt: null }, take: 1, select: { id: true } },
    },
    orderBy: { name: "asc" },
  });

  const opciones: OpcionDeRecipiente[] = [];
  for (const e of equipos) {
    if (!(await puedeVerEquipo(userAccountId, e))) continue;
    const { libreYSano, motivos } = clasificar({
      id: e.id,
      lifecycleStatus: e.lifecycleStatus,
      condicion: e.conditionReports[0]?.condition ?? null,
      enUso: e.fermentationRuns.length > 0,
    });
